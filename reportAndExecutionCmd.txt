For allure:

//to override the existing reporter in config and generate allure report provide below command:
npx playwright test --grep "@Web" --reporter=line,allure-playwright

//to generate allure report from allure results folder:
allure generate ./allure-results --clean

//to open report
allure open ./allure-report


//to run with different config file and project
npx playwright test tests/MoreValidations.spec.js --config playwright.config1.js --project=chrome

cucumber:

//to run cucumber test:
npx cucumber-js --exit

//to run specific feature file:
npx cucumber-js features/ErrorValidation.feature --exit

//to run with tags:
npx cucumber-js --tags "@Regression" --exit

//to run scenarios parallelly at a time:
npx cucumber-js features/EndToEndFlow.feature --parallel 2 --exit

//to generate the html report for cucumber:
npx cucumber-js features/EndToEndFlow.feature --parallel 2 --exit --format html:cucumber-report.html

//to run scenarios by adding run command in package.json:
"cucumberRegression": "npx cucumber-js --tags \"@Regression\" --retry 1 --exit --format html:cucumber-report.html"
